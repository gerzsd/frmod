 18def get_freq_ratios_classic(vr,
 19                            mask,
 20                            binc=20,
 21                            nodata=-99999.,
 22                            categorical=False,
 23                            ls_marker=1,
 24                            nls_marker=0):
 25    """
 26    Get the frequency ratio of the landslide parts and the whole area.
 27
 28    Parameters
 29    ----------
 30    vr : Array
 31        The array of the analyzed variable. Numeric values.
 32    mask : Array
 33        The array of the mask.
 34    binc : int, optional
 35        Bin count for the histogram of the non-categorical variables.
 36        The default is 20.
 37    nodata : int / float, optional
 38        The nodata value of the vr grid.
 39        The default is -99999..
 40    categorical : bool, optional
 41        Set true if the analysed variable raster is categorical.
 42        Categories must be marked with unique integers in the rasters.
 43        The dafault is False.
 44    ls_marker : int, optional
 45        The value marking the landslide parts.
 46        The default is 1.
 47    nls_marker : int, optional
 48        The value marking the non-landslide parts.
 49        The default is 0.
 50
 51    Returns
 52    -------
 53    frequency_ratios : Array
 54        The frequency ratio values. Length: number of bins.
 55    hst_bins : Array
 56        Array containing the edges of the bins. Length: number of bins + 1.
 57    fr_stat_df : pandas.DataFrame
 58        DataFrame with the statistics.
 59        Columns:
 60            'min', 'max', 'LS_density', 'NLS_density', 'frequency_ratio'.
 61
 62    """
 63    vr = np.ma.masked_where((vr == nodata), vr)
 64    in_all = np.logical_or(mask == ls_marker, mask == nls_marker)
 65    ls_area = np.count_nonzero(mask == ls_marker)
 66    all_area = np.count_nonzero(mask == nls_marker) + ls_area
 67    if categorical:
 68        bin_edges = np.unique(vr[in_all].compressed())
 69        bin_edges = np.append(bin_edges, bin_edges[-1] + 1)
 70        ls_hst = np.histogram(vr[mask == ls_marker],
 71                              bins=bin_edges,
 72                              density=False)
 73        all_hst = np.histogram(vr[in_all],
 74                               bins=bin_edges,
 75                               density=False)
 76    else:
 77        glob_lim = (vr[in_all].min(),
 78                    vr[in_all].max())
 79        ls_hst = np.histogram(vr[mask == ls_marker], bins=binc,
 80                              range=glob_lim, density=False)
 81        all_hst = np.histogram(vr[in_all], bins=binc,
 82                               range=glob_lim, density=False)
 83
 84    # Histogram density for the landslide part
 85    ls_hst_d = ls_hst[0] / ls_area
 86    # Histogram density for the non-landslide part
 87    all_hst_d = all_hst[0] / all_area
 88    frequency_ratios = ls_hst_d / all_hst_d
 89
 90    hst_bins = all_hst[1]
 91    mn = hst_bins[:-1]
 92    mx = hst_bins[1:]
 93
 94    # Create a pd.DataFrame for the bins, densities, and the frequency ratio
 95    data = [mn, mx, ls_hst_d, all_hst_d, frequency_ratios]
 96    # columns = ["Min", "Max", "LS_density", "NLS_density", "frequency_ratio"]
 97    data = {'min': mn,
 98            'max': mx,
 99            'LS_density': ls_hst_d,
100            'NLS_density': all_hst_d,
101            'frequency_ratio': frequency_ratios}
102    fr_stat_df = pd.DataFrame(data=data)
103    return frequency_ratios, hst_bins, fr_stat_df

105def get_freq_ratios(vr,
106                    mask,
107                    binc=100,
108                    nodata=-9999.,
109                    categorical=False,
110                    normalize=False,
111                    ls_marker=1,
112                    nls_marker=0):
113    """
114    Get the frequency ratio of the landslide and non-landslide parts.
115
116    Parameters
117    ----------
118    vr : Array
119        The array of the analyzed variable. Numeric values.
120    mask : Array
121        The array of the mask.
122    binc : int, optional
123        Bin count for the histogram of the non-categorical variables.
124        The default is 100.
125    categorical : bool, optional
126        Set true if the analysed variable raster is categorical.
127        Categories must be marked with unique integers in the rasters.
128        The dafault is False.
129    normalize : bool, optional
130        Set True for normalized weights (0, 1)
131        The default is False.
132    ls_marker : int, optional
133        The value marking the landslide parts.
134        The default is 1.
135    nls_marker : int, optional
136        The value marking the non-landslide parts.
137        The default is 0.
138
139    Returns
140    -------
141    frequency_ratios : Array
142        The frequency ratio values. Length: number of bins.
143    hst_bins : Array
144        Array containing the edges of the bins. Length: number of bins + 1.
145    fr_stat_df : pandas.DataFrame
146        DataFrame with the statistics.
147
148    """
149    ls_area = np.count_nonzero(mask == ls_marker)
150    nls_area = np.count_nonzero(mask == nls_marker)
151    if categorical:
152        bin_edges = np.unique(np.ma.compressed(vr[(vr != nodata)]))
153        bin_edges = np.append(bin_edges, bin_edges[-1] + 1)
154        ls_hst = np.histogram(vr[mask == ls_marker],
155                              bins=bin_edges,
156                              density=False)
157        nls_hst = np.histogram(vr[mask == nls_marker],
158                               bins=bin_edges,
159                               density=False)
160    else:
161        glob_lim = (vr[vr != nodata].min(),
162                    vr[vr != nodata].max())
163        ls_hst = np.histogram(vr[mask == ls_marker], bins=binc,
164                              range=glob_lim, density=False)
165        nls_hst = np.histogram(vr[mask == nls_marker], bins=binc,
166                               range=glob_lim, density=False)
167    # Histogram density for the landslide part
168    ls_hst_d = ls_hst[0] / ls_area
169    # Histogram density for the non-landslide part
170    nls_hst_d = nls_hst[0] / nls_area
171    # Histogram bins
172    hst_bins = ls_hst[1]
173    mn = hst_bins[:-1]
174    mx = hst_bins[1:]
175    # Frequency ratios
176    fr = ls_hst_d / nls_hst_d
177    if normalize:
178        frequency_ratios = (fr - fr.min()) / (fr.max() - fr.min())
179    else:
180        frequency_ratios = fr
181    # Create a pd.DataFrame for the bins, densities, and the frequency ratio
182    data = [mn, mx, ls_hst_d, nls_hst_d, frequency_ratios]
183    # columns = ["Min", "Max", "LS_density", "NLS_density", "frequency_ratio"]
184    data = {'min': mn,
185            'max': mx,
186            'LS_density': ls_hst_d,
187            'NLS_density': nls_hst_d,
188            'frequency_ratio': fr}
189    fr_stat_df = pd.DataFrame(data=data)
190    return frequency_ratios, hst_bins, fr_stat_df

193def reclass_raster(vr, f_ratios, bin_edges, verbose=False,
194                   categorical=False, nodata=None):
195    """
196    Create an array with the frequency ratios.
197
198    Parameters
199    ----------
200    vr : Array
201        Array of the analysed variable to be reclassified.
202    f_ratios : Array
203        The frequency ratio values.
204        Length: number of bins.
205    bin_edges : Array
206        Array containing the edges of the bins.
207        Length: number of bins + 1.
208    verbose : bool
209        Set True to print the bin ranges and reclass values.
210    categorical : bool, optional
211        Set True if vr holds category IDs. A cell is then reclassified only
212        if its value is exactly one of the categories in bin_edges, so
213        categories that were not part of the analysis stay nodata.
214        The default is False.
215    nodata : int / float, optional
216        The nodata value of vr. Cells with this value are not reclassified,
217        even if the value falls inside the range of the bins.
218        The default is None (no nodata value).
219
220    Returns
221    -------
222    reclassed : Array
223        Reclassified array with the appropriate frequency ratio values.
224        Cells that were not reclassified are set to -99999.
225
226    """
227    reclassed = np.ones(vr.shape) * -99999
228    last_bin = len(f_ratios) - 1
229    for i in range(0, len(f_ratios)):
230        # Reclassifying the raster by assigning the frequency ratio
231        # values of each bin to the corresponding raster values.
232        mn = bin_edges[:-1][i]
233        mx = bin_edges[1:][i]
234        vrange = mx - mn
235        if categorical:
236            # The bin of a category starts at the category ID.
237            to_reclass = (vr == mn)
238        elif i == last_bin:
239            # The last bin includes its upper edge, like in np.histogram.
240            to_reclass = (vr >= mn) & (vr <= mx)
241        else:
242            to_reclass = (vr >= mn) & (vr < mx)
243        if nodata is not None:
244            to_reclass &= (vr != nodata)
245        reclassed[to_reclass] = f_ratios[i]
246        if verbose:
247            print("Min: {} Max: {} Range: {} Ratio: {}".format(
248                mn, mx, vrange, f_ratios[i])
249                 )
250    return reclassed

363    def __init__(self, name, path, ls_marker=1, nls_marker=0, fold_count=5,
364                 use_gdal=False):
365        """
366        Create a LandslideMask object.
367
368        Parameters
369        ----------
370        name : str
371            Name of the landslide mask.
372        path : str
373            Path to the file used as the mask in the analysis.
374            A SAGA grid (.sdat) by default.
375            With use_gdal=True: any GDAL-compatible raster file.
376        ls_marker : int, optional
377            Value marking the landslide pixels.
378            The default is 1.
379        nls_marker : int, optional
380            Value marking the non-landslide pixels.
381            Must be different from the nodata value.
382            The default is 0.
383        fold_count : int, optional
384            The number of cross validation folds. The default is 5.
385        use_gdal : bool, optional
386            False: the SAGA grid is read without GDAL.
387            True: GDAL reads the raster, it must be installed.
388            The default is False.
389
390        Returns
391        -------
392        None.
393
394        """
395        # Name of the LandslideMask
396        self.name = name
397        # Path to the input mask file.
398        self.path = path
399        # Read the input mask file with GDAL?
400        self.use_gdal = use_gdal
401        # 2D array representation of the input mask file and its NoData value.
402        self.grid, self.nodata = utils.read_raster(path, use_gdal=use_gdal)
403        # Value marking the landslide cells.
404        self.ls_marker = ls_marker
405        # Value marking the non-landslide cells. Different from nodata!
406        self.nls_marker = nls_marker
407        # Number of folds
408        self.fold_count = fold_count
409        # (fold_count no. of (2, n) size arrays) with the split positions
410        self.split_locations = self.get_splits(folds=self.fold_count)
411        # List of (train_areas, valid_positions) ndarray pairs.
412        self.t_v = [self.get_train_area(a) for a in self.split_locations]
413        # Training areas (1), validation cells count as non-landslide.
414        # Input mask for get_freq_ratios.
415        self.train_areas = [t_v[0] for t_v in self.t_v]
416        # Positions of the validation cells.
417        self.valid_positions = [t_v[1] for t_v in self.t_v]
418
419    def get_splits(self, folds=5):
420        """
421        Split the mask for cross validation.
422
423        Parameters
424        ----------
425        folds : TYPE, optional
426            Number of desired folds. The default is 5.
427
428        Returns
429        -------
430        split_locations : list of ndarrays
431            (fold_count no. of (2, n) size arrays) with the split positions.
432
433        """
434        valid = np.array(np.nonzero(self.grid == self.ls_marker))
435        valid_transposed = valid.T
436        np.random.shuffle(valid_transposed)
437        valid_transposed_split = np.array_split(valid_transposed, folds)
438        split_locations = [i.T for i in valid_transposed_split]
439        return split_locations
440
441    def get_train_area(self, split_to_omit):
442        """
443        Get the train_area grid and the validation cell positions.
444
445        Parameters
446        ----------
447        split_to_omit : list
448            List of the position of the validation cells. It is used
449            to construct the valid_position array. Cells marked here
450            are turned into non-landslide cells.
451
452        Returns
453        -------
454        train_area : list
455            List of train area grids. Similar format to the self.grid.

457            Lists the positions of the validation cells for the folds.
458            Positions are given as two arrays:
459                1st: row index increasing from top to bottom.
460                2nd: column index increasing from left to right.
461
462        """
463        train_area = np.copy(self.grid)
464        valid_position = (split_to_omit[0], split_to_omit[1])
465        train_area[valid_position] = self.nls_marker
466        return train_area, valid_position

486class FRAnalysis():
487    """Frequency ratio analysis of a LandslideMask and a list of VRasters.
488
489    The steps of the analysis:
490
491    1. run_analysis() is run upon instantiation for each fold of the
492        ls_mask and each VRaster. This yields the frequency ratio statistics
493        and the reclassified grids.
494    2. get_result() Compute the susceptibility grid and other related data.
495    3. get_src() Compute the success rate curves.
496        The success rate curve is the cumulative frequency distribution
497        of the landslide cells in the susceptibility categories.
498        Computed for each fold.
499    4. get_auc() Compute the area under the success rate curves for scoring.
500        Computed for each fold, then averaged for an overall score.
501        Smaller values are better.
502    """
503
504    def __init__(self, ls_mask, var_list, classic_mode=True):
505        """
506        Create the FRAnalysis object.
507
508        Create the FRAnalysis object and its instanced variables.
509        The last step of the instantiation is creating the rc_folds list.
510        The rc_folds holds the reclassified VRaster.grid-s for the folds 
511        of the ls_mask for each VRaster in the var_list.
512
513        Parameters
514        ----------
515        ls_mask : LandslideMask
516            The LandslideMask object for the analysis.
517        var_list : list
518            List of VRaster objects to be analyzed.
519        classic_mode : bool
520            True: get frequency ratios with get_freq_ratios_classic
521            False: get likelihood ratios with get_freq_ratios
522
523        Returns
524        -------
525        None.
526
527        """
528        # Mode
529        self.classic_mode=classic_mode
530        # Input LandslideMask
531        self.ls_mask = ls_mask
532        # List of input VRasters
533        self.var_list = var_list
534        # Number of VRasters
535        self.var_count = len(var_list)
536
537        # AUC, area under the success rate curve
538        # Sum of the success_rates of the folds
539        self.auc_folds = []
540        # Mean AUC
541        self.auc_mean = None
542        # Standard deviations of the AUCs
543        self.auc_std = None
544
545        # Percentile bins of the fresult
546        self.ranks = None
547        # Fold_susceptibility, the susceptibility grids of the folds
548        self.fold_susceptibility = None
549        self.fold_percentiles = []
550        # Final result, the mean estimated susceptibility map over the folds.
551        self.fresult = None
552        # Final result in the percentile form
553        self.percentile_grid = None
554
555        # Susceptibility value bins for the percentiles
556        self.valid_perc = []
557
558        # The distributions of % ranks for the validation cells.
559        self.v_dist = []
560        # Bins for the above v_dist for conversion to percentiles
561        self.v_bins = []
562
563        # Frequency ratio analysis results for each VRaster and fold.
564        self.stats = {}
565
566        # Frequency ratio analysis results for each VRaster and fold.
567        # keyword: VRaster.name
568        # value: list of pd.DataFrames, 1 DF / fold
569        self.fr_stats_full = {}
570
571        # List of success rates for the folds.
572        self.success_rates = None
573        # pandas DataFrame of the success rates for the folds.
574        self.src_df = None
575
576        # Reclassified grids for each variable and fold
577        # Shape: [var_count, ls_mask.fold_count, rows, columns]
578        self.rc_folds = [self.run_analysis(vraster, self.ls_mask)
579                         for vraster in self.var_list]
580
581    def run_analysis(self, vrr, lsm):
582        """
583        Frequency ratio analysis of one VRaster and the LandslideMask folds.
584
585        Parameters
586        ----------
587        vrr : VRaster
588            Supplies the grid and other parameters for the
589            get_freq_ratios function.
590        lsm : LandslideMask
591            Supplies the input masks for the get_freq_ratios functions.
592
593        Returns
594        -------
595        rc_folds : ndarray
596            The reclassified grids after the frequency ratio analysis of the
597            VRaster and each fold of the LandslideMask.
598            Shape: (self.var_count, lsm.fold_count, vrr.shape[0], vrr.shape[1])
599
600        """
601        # Array for storing the reclassified VRaster.grids for the folds
602        rc_folds = []
603        # Create arrays for the statistics
604        all_frq_ratios = []
605        all_hst_bins = []
606        all_folds_statistics = []
607
608        # Run the analysis with the training areas of the different folds
609        # Iterating over the train_areas of the folds
610        for msk in lsm.train_areas:
611            if self.classic_mode:
612                fr_data = get_freq_ratios_classic(vr=vrr.grid,
613                                                  mask=msk,
614                                                  binc=vrr.bins,
615                                                  nodata=vrr.nodata,
616                                                  categorical=vrr.categorical,
617                                                  ls_marker=lsm.ls_marker,
618                                                  nls_marker=lsm.nls_marker)
619            else:
620                fr_data = get_freq_ratios(vr=vrr.grid,
621                                          mask=msk,
622                                          binc=vrr.bins,
623                                          nodata=vrr.nodata,
624                                          categorical=vrr.categorical,
625                                          normalize=False,
626                                          ls_marker=lsm.ls_marker,
627                                          nls_marker=lsm.nls_marker
628                                          )
629            frq_ratios = fr_data[0]
630            hst_bins = fr_data[1]
631            # pd.DataFrame with the densities and the frequency ratios
632            all_folds_statistics.append(fr_data[2])
633            # Prepare the statistics DataFrame
634            all_frq_ratios.append(frq_ratios)
635            all_hst_bins.append(hst_bins)
636            # Reclassify the VRaster.grid
637            reclassed = reclass_raster(vrr.grid, frq_ratios, hst_bins,
638                                       categorical=vrr.categorical,
639                                       nodata=vrr.nodata)
640            # Append the reclassified VRaster.grid to rc_folds
641            rc_folds.append(reclassed)
642
643        # Adding all_folds_statistics DF to the fr_stats_full dict
644        self.fr_stats_full[vrr.name] = all_folds_statistics

657    def get_result(self, product=False):
658        """
659        Get the susceptibility estimation and ranks.
660        
661        Parameters
662        ----------
663        product : bool, optional
664            If true: the final susceptibility is calculated as the product
665            of the susceptibility grids of the folds.
666            If false: the final susceptibility is calculated as the average
667            of the susceptibility grids of the folds.
668            Default is false.
669
670        Returns
671        -------
672        self.ranks : ndarray
673            Numpy array (101,) with the percentile bins.
674
675        """
676        rc_folds = self.rc_folds
677        valid_positions = self.ls_mask.valid_positions
678        # Start from empty lists, so calling get_result() again
679        # does not add the folds a second time.
680        self.valid_perc = []
681        self.v_dist = []
682        self.v_bins = []
683        self.fold_percentiles = []
684        result = []
685        # TODO remove percentile_result if really not used
686        percentile_result = []
687        # percentile bins
688        percentile_bins = [x * 0.01 for x in range(0, 101)]
689        # Iterate over the folds of the ls_mask.
690        for i in range(0, self.ls_mask.fold_count):
691            if product:
692                fold_result = np.ones(rc_folds[0][0].shape)
693            else:
694                fold_result = np.zeros(rc_folds[0][0].shape)
695            # Cells that have no valid value in at least one of the layers.
696            # Valid frequency ratios are >= 0, nodata is -99999.
697            nodata_cells = np.zeros(fold_result.shape, dtype=bool)
698
699            # Iterate over the reclassified grids of fold #i
700            for j in rc_folds:
701                nodata_cells |= ~(j[i] >= 0)
702                if product:
703                    fold_result *= j[i]
704                else:
705                    # Average the reclassified  grids of fold #i.
706                    fold_result += j[i] / self.var_count
707            # Without this, nodata can look valid after the multiplication,
708            # eg. (-99999) * (-99999) is a large positive number.
709            fold_result[nodata_cells] = -99999
710
711            # Percentile bin edges for conversion to percentiles
712            valid_perc = np.quantile(fold_result[fold_result >= 0],
713                                     percentile_bins,
714                                     method='nearest')
715
716            self.valid_perc.append(valid_perc)
717            v_to_score = fold_result[valid_positions[i]]
718            v_dist, v_bins = np.histogram(v_to_score,
719                                          valid_perc,
720                                          density=False)
721
722            v_dist = np.array(v_dist)
723            v_dist = v_dist / sum(v_dist)
724            # Append the probability density function to vdist
725            self.v_dist.append(v_dist)
726            self.v_bins.append(v_bins)
727            result.append(fold_result)
728
729        # fresult: cell by cell average of the result array
730        fresult = sum(result) / self.ls_mask.fold_count
731
732        # Assign the results from the validation folds to the location
733        # of the landslide mask and create the percentile grids for the folds
734        for i in range(0, self.ls_mask.fold_count):
735            fresult[valid_positions[i]] = result[i][valid_positions[i]]
736            result[i][result[i] < 0] = -99999
737            fp = reclass_raster(result[i],
738                                percentile_bins[1:],
739                                self.valid_perc[i])
740            fp[fp >= 0] = fp[fp >= 0] * 100
741            self.fold_percentiles.append(fp)
742
743        # Set all invalid values (<0) to -99999 (nodata)
744        fresult[fresult < 0] = -99999
745
746        self.fold_susceptibility = result
747        self.fresult = fresult
748        self.ranks = np.quantile(fresult[fresult >= 0], percentile_bins)
749        return self.ranks

194def reclass_raster(vr, f_ratios, bin_edges, verbose=False,
195                   categorical=False, nodata=None):
196    """
197    Create an array with the frequency ratios.
198
199    Parameters
200    ----------
201    vr : Array
202        Array of the analysed variable to be reclassified.
203    f_ratios : Array
204        The frequency ratio values.
205        Length: number of bins.
206    bin_edges : Array
207        Array containing the edges of the bins.
208        Length: number of bins + 1.
209    verbose : bool
210        Set True to print the bin ranges and reclass values.
211    categorical : bool, optional
212        Set True if vr holds category IDs. A cell is then reclassified only
213        if its value is exactly one of the categories in bin_edges, so
214        categories that were not part of the analysis stay nodata.
215        The default is False.
216    nodata : int / float, optional
217        The nodata value of vr. Cells with this value are not reclassified,
218        even if the value falls inside the range of the bins.
219        The default is None (no nodata value).
220
221    Returns
222    -------
223    reclassed : Array
224        Reclassified array with the appropriate frequency ratio values.
225        Cells that were not reclassified are set to -99999.
226
227    """
228    reclassed = np.ones(vr.shape) * -99999
229    last_bin = len(f_ratios) - 1
230    for i in range(0, len(f_ratios)):
231        # Reclassifying the raster by assigning the frequency ratio
232        # values of each bin to the corresponding raster values.
233        mn = bin_edges[:-1][i]
234        mx = bin_edges[1:][i]
235        vrange = mx - mn
236        if categorical:
237            # The bin of a category starts at the category ID.
238            to_reclass = (vr == mn)
239        elif i == last_bin:
240            # The last bin includes its upper edge, like in np.histogram.
241            to_reclass = (vr >= mn) & (vr <= mx)
242        else:
243            to_reclass = (vr >= mn) & (vr < mx)
244        if nodata is not None:
245            to_reclass &= (vr != nodata)
246        reclassed[to_reclass] = f_ratios[i]
247        if verbose:
248            print("Min: {} Max: {} Range: {} Ratio: {}".format(
249                mn, mx, vrange, f_ratios[i])
250                 )
251    return reclassed

661

663        ----------
664        product : bool, optional
665            If true: the final susceptibility is calculated as the product
666            of the susceptibility grids of the folds.
667            If false: the final susceptibility is calculated as the average
668            of the susceptibility grids of the folds.
669            Default is false.
670
671        Returns
672        -------
673        self.ranks : ndarray
674            Numpy array (101,) with the percentile bins.
675
676        """
677        rc_folds = self.rc_folds
678        valid_positions = self.ls_mask.valid_positions
679        # Start from empty lists, so calling get_result() again
680        # does not add the folds a second time.
681        self.valid_perc = []
682        self.v_dist = []
683        self.v_bins = []
684        self.fold_percentiles = []
685        result = []
686        # TODO remove percentile_result if really not used
687        percentile_result = []
688        # percentile bins
689        percentile_bins = [x * 0.01 for x in range(0, 101)]
690        # Iterate over the folds of the ls_mask.
691        for i in range(0, self.ls_mask.fold_count):
692            if product:
693                fold_result = np.ones(rc_folds[0][0].shape)
694            else:
695                fold_result = np.zeros(rc_folds[0][0].shape)
696            # Cells that have no valid value in at least one of the layers.
697            # Valid frequency ratios are >= 0, nodata is -99999.
698            nodata_cells = np.zeros(fold_result.shape, dtype=bool)
699
700            # Iterate over the reclassified grids of fold #i
701            for j in rc_folds:
702                nodata_cells |= ~(j[i] >= 0)
703                if product:
704                    fold_result *= j[i]
705                else:

The steps of the analysis:

  1. run_analysis() is run upon instantiation for each fold of the ls_mask and each VRaster. This yields the frequency ratio statistics and the reclassified grids.
  2. get_result() Compute the susceptibility grid and other related data.
  3. get_src() Compute the success rate curves. The success rate curve is the cumulative frequency distribution of the landslide cells in the susceptibility categories. Computed for each fold.
